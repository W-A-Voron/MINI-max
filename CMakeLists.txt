cmake_minimum_required(VERSION 3.20)
project(minimax VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(MINIMAX_BUILD_SERVER "Build the C server" ON)
option(MINIMAX_BUILD_CLIENT "Build the Qt6 client" ON)
option(MINIMAX_SANITIZE "Enable ASan/UBSan" OFF)

if(MINIMAX_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(NOT MSVC)
  add_compile_options(-Wall -Wextra)
endif()

enable_testing()
add_subdirectory(shared)
if(MINIMAX_BUILD_SERVER)
  add_subdirectory(server)
endif()
if(MINIMAX_BUILD_CLIENT)
  enable_language(CXX)
  add_subdirectory(client)
endif()
